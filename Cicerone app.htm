// App State

let activePool = [];

let currentIndex = 0;

let userStats = JSON.parse(localStorage.getItem('cicerone_stats')) || { total: 0, correct: 0, bookmarks: [] };

 

function saveStats() {

    localStorage.setItem('cicerone_stats', JSON.stringify(userStats));

    updateDashboardUI();

}

 

function updateDashboardUI() {

    document.getElementById('stat-total').innerText = userStats.total;

    const acc = userStats.total > 0 ? Math.round((userStats.correct / userStats.total) * 100) : 0;

    document.getElementById('stat-accuracy').innerText = acc + '%';

    document.getElementById('stat-bookmarks').innerText = userStats.bookmarks.length;

}

 

function showDashboard() {

    document.getElementById('dashboard-view').classList.remove('hidden');

    document.getElementById('quiz-view').classList.add('hidden');

    document.getElementById('home-btn').classList.add('hidden');

    updateDashboardUI();

}

 

function startQuiz(count) {

    if (count === 'bookmarks') {

        activePool = questionBank.filter(q => userStats.bookmarks.includes(q.id));

        if (activePool.length === 0) {

            alert('No bookmarked questions available!');

            return;

        }

    } else {

        // Expand bank dynamically to requested length by shuffling existing set

        let pool = [...questionBank];

        while (pool.length < count && pool.length > 0) {

            pool = pool.concat(questionBank);

        }

        activePool = pool.sort(() => 0.5 - Math.random()).slice(0, count);

    }

 

    currentIndex = 0;

    document.getElementById('dashboard-view').classList.add('hidden');

    document.getElementById('quiz-view').classList.remove('hidden');

    document.getElementById('home-btn').classList.remove('hidden');

    loadQuestion();

}

 

function loadQuestion() {

    const q = activePool[currentIndex];

    document.getElementById('q-progress').innerText = `Question ${currentIndex + 1}/${activePool.length}`;

    document.getElementById('q-text').innerText = q.q;

   

    // Update Bookmark state

    const isBookmarked = userStats.bookmarks.includes(q.id);

    const bmBtn = document.getElementById('bookmark-toggle');

    bmBtn.classList.toggle('active', isBookmarked);

 

    // Load options

    const container = document.getElementById('options-container');

    container.innerHTML = '';

    q.opts.forEach((opt, idx) => {

        const btn = document.createElement('button');

        btn.className = 'opt-btn';

        btn.innerText = opt;

        btn.onclick = () => handleAnswer(idx);

        container.appendChild(btn);

    });

 

    document.getElementById('feedback-box').classList.add('hidden');

    document.getElementById('next-btn').classList.add('hidden');

}

 

function handleAnswer(selectedIdx) {

    const q = activePool[currentIndex];

    const buttons = document.querySelectorAll('.opt-btn');

   

    buttons.forEach((btn, idx) => {

        btn.disabled = true;

        if (idx === q.ans) btn.classList.add('correct');

        if (idx === selectedIdx && selectedIdx !== q.ans) btn.classList.add('incorrect');

    });

 

    userStats.total++;

    if (selectedIdx === q.ans) userStats.correct++;

    saveStats();

 

    document.getElementById('feedback-text').innerText = q.exp;

    document.getElementById('feedback-box').classList.remove('hidden');

    document.getElementById('next-btn').classList.remove('hidden');

}

 

function toggleBookmark() {

    const q = activePool[currentIndex];

    const idx = userStats.bookmarks.indexOf(q.id);

    if (idx > -1) {

        userStats.bookmarks.splice(idx, 1);

    } else {

        userStats.bookmarks.push(q.id);

    }

    saveStats();

    document.getElementById('bookmark-toggle').classList.toggle('active', idx === -1);

}

 

function nextQuestion() {

    currentIndex++;

    if (currentIndex < activePool.length) {

        loadQuestion();

    } else {

        alert(`Quiz Complete! Final Score: ${userStats.correct} correct out of ${userStats.total} total overall attempts.`);

        showDashboard();

    }

}
